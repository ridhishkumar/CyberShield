cmake_minimum_required(VERSION 3.18)
project(CyberShield LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
find_package(CURL REQUIRED)
include(CTest)
add_library(sqlite STATIC vendor/sqlite/sqlite3.c)
target_include_directories(sqlite PUBLIC vendor/sqlite)
add_library(scanner ../local-scanner/src/scanner.cpp)
target_compile_definitions(scanner PUBLIC ASIO_STANDALONE)
target_include_directories(scanner PUBLIC ../local-scanner/include vendor/asio/include)
add_library(website-inspector src/website_inspector.cpp src/website_tld_policy.cpp)
target_include_directories(website-inspector PUBLIC include vendor/json vendor/sqlite)
target_link_libraries(website-inspector PUBLIC CURL::libcurl OpenSSL::Crypto scanner Threads::Threads)
add_executable(cybershield src/main.cpp src/database.cpp src/auth.cpp src/detectors.cpp src/account.cpp src/workspace.cpp)
target_sources(cybershield PRIVATE src/integrations.cpp src/local_worker.cpp src/selected_file.cpp src/windows_verdict.cpp src/dns.cpp)
target_link_libraries(cybershield PRIVATE CURL::libcurl website-inspector)
if(MSVC)
 target_compile_options(cybershield PRIVATE /utf-8 /bigobj)
 target_compile_options(website-inspector PRIVATE /utf-8 /bigobj)
endif()
target_include_directories(cybershield PRIVATE include vendor/crow/include vendor/json)
target_compile_definitions(cybershield PRIVATE CROW_DISABLE_STATIC_DIR)
target_link_libraries(cybershield PRIVATE sqlite scanner OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
add_executable(cybershield-scanner ../local-scanner/src/main.cpp)
target_link_libraries(cybershield-scanner PRIVATE scanner Threads::Threads)
if(BUILD_TESTING)
 add_executable(website-inspector-tests tests/website_inspector_tests.cpp)
 target_include_directories(website-inspector-tests PRIVATE include vendor/json)
 target_link_libraries(website-inspector-tests PRIVATE website-inspector)
 if(MSVC)
  target_compile_options(website-inspector-tests PRIVATE /utf-8 /bigobj)
 endif()
 if(WIN32)
  target_link_libraries(website-inspector-tests PRIVATE ws2_32 mswsock)
 endif()
 add_test(NAME website-inspector-tests COMMAND website-inspector-tests)
 add_executable(software-detector-tests tests/software_detector_tests.cpp src/detectors.cpp)
 target_include_directories(software-detector-tests PRIVATE include vendor/crow/include vendor/json)
 target_link_libraries(software-detector-tests PRIVATE website-inspector)
 if(MSVC)
  target_compile_options(software-detector-tests PRIVATE /utf-8 /bigobj)
 endif()
 if(WIN32)
  target_link_libraries(software-detector-tests PRIVATE ws2_32 mswsock)
 endif()
 add_test(NAME software-detector-tests COMMAND software-detector-tests)
endif()
if(WIN32)
 target_compile_definitions(scanner PUBLIC _WIN32_WINNT=0x0A00 NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(scanner PUBLIC ws2_32 mswsock)
 target_link_libraries(cybershield PRIVATE ole32 uuid)
endif()
