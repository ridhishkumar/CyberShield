cmake_minimum_required(VERSION 3.18)
project(CyberShield LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
find_package(CURL REQUIRED)
add_library(sqlite STATIC vendor/sqlite/sqlite3.c)
target_include_directories(sqlite PUBLIC vendor/sqlite)
add_library(scanner ../local-scanner/src/scanner.cpp)
target_compile_definitions(scanner PUBLIC ASIO_STANDALONE)
target_include_directories(scanner PUBLIC ../local-scanner/include vendor/asio/include)
add_executable(cybershield src/main.cpp src/database.cpp src/auth.cpp src/detectors.cpp src/account.cpp src/workspace.cpp)
target_sources(cybershield PRIVATE src/integrations.cpp src/local_worker.cpp src/selected_file.cpp src/windows_verdict.cpp src/dns.cpp)
target_link_libraries(cybershield PRIVATE CURL::libcurl)
if(MSVC)
 target_compile_options(cybershield PRIVATE /utf-8 /bigobj)
endif()
target_include_directories(cybershield PRIVATE include vendor/crow/include vendor/json)
target_compile_definitions(cybershield PRIVATE CROW_DISABLE_STATIC_DIR)
target_link_libraries(cybershield PRIVATE sqlite scanner OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
add_executable(cybershield-scanner ../local-scanner/src/main.cpp)
target_link_libraries(cybershield-scanner PRIVATE scanner Threads::Threads)
if(WIN32)
 target_compile_definitions(scanner PUBLIC _WIN32_WINNT=0x0A00 NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(scanner PUBLIC ws2_32 mswsock)
 target_link_libraries(cybershield PRIVATE ole32 uuid)
endif()
